cmake_minimum_required(VERSION 3.24)
project(docscan VERSION 0.1.0 LANGUAGES C CXX)
include(CTest)
include(GNUInstallDirs)
include(CMakePackageConfigHelpers)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(DOCSCAN_BUILD_TESTS "Build deterministic native tests" ON)
option(DOCSCAN_BUILD_EXAMPLES "Build the in-process CLI example" ON)
set(ONNXRUNTIME_ROOT "" CACHE PATH "Official ONNX Runtime native distribution")
find_package(OpenCV 4.14 REQUIRED COMPONENTS core imgproc imgcodecs)
find_path(ORT_INCLUDE onnxruntime_cxx_api.h HINTS "${ONNXRUNTIME_ROOT}/include" REQUIRED)
find_library(ORT_LIBRARY NAMES onnxruntime HINTS "${ONNXRUNTIME_ROOT}/lib" REQUIRED)
add_library(ORT::Runtime SHARED IMPORTED)
set_target_properties(ORT::Runtime PROPERTIES INTERFACE_INCLUDE_DIRECTORIES "${ORT_INCLUDE}")
if(WIN32)
    find_file(ORT_DLL onnxruntime.dll HINTS "${ONNXRUNTIME_ROOT}/lib" REQUIRED)
    set_target_properties(ORT::Runtime PROPERTIES IMPORTED_IMPLIB "${ORT_LIBRARY}" IMPORTED_LOCATION "${ORT_DLL}")
else()
    set_target_properties(ORT::Runtime PROPERTIES IMPORTED_LOCATION "${ORT_LIBRARY}")
endif()
add_library(docscan SHARED src/api.cpp src/pipeline.cpp src/geometry.cpp src/filters.cpp src/paper-restoration.cpp src/paper-field.cpp src/quality.cpp src/appearance-plan.cpp src/opencv-raster.cpp src/models.cpp src/corner-refinement.cpp src/shadow-model.cpp src/adaptive-enhancement.cpp src/pdf.cpp)
add_library(docscan::docscan ALIAS docscan)
target_include_directories(docscan
    PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include>
    PRIVATE ${OpenCV_INCLUDE_DIRS})
target_compile_definitions(docscan PUBLIC DOCSCAN_SHARED PRIVATE DOCSCAN_BUILD ORT_API_MANUAL_INIT)
target_link_libraries(docscan PRIVATE ${OpenCV_LIBS} ORT::Runtime)
if(MSVC)
    target_compile_options(docscan PRIVATE /W4 /permissive- /utf-8 /fp:precise)
else()
    target_compile_options(docscan PRIVATE -Wall -Wextra -Wpedantic -fvisibility=hidden)
endif()
set_target_properties(docscan PROPERTIES VERSION ${PROJECT_VERSION} SOVERSION 1)
if(WIN32)
    add_custom_command(TARGET docscan POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different
        $<TARGET_RUNTIME_DLLS:docscan> $<TARGET_FILE_DIR:docscan> COMMAND_EXPAND_LISTS)
endif()
if(DOCSCAN_BUILD_EXAMPLES)
    add_executable(docscan_cli examples/scan.cpp)
    target_link_libraries(docscan_cli PRIVATE docscan)
    install(TARGETS docscan_cli RUNTIME DESTINATION bin)
endif()
if(DOCSCAN_BUILD_TESTS)
    add_executable(corner_contract tests/corner_contract.cpp src/corner-refinement.cpp)
    target_include_directories(corner_contract PRIVATE src include ${OpenCV_INCLUDE_DIRS})
    target_link_libraries(corner_contract PRIVATE ${OpenCV_LIBS})
    add_test(NAME corner_contract COMMAND corner_contract)
    add_executable(abi_contract tests/abi_contract.cpp)
    target_link_libraries(abi_contract PRIVATE docscan)
    add_test(NAME abi_contract COMMAND abi_contract)
    add_executable(c_api_smoke tests/c_api_smoke.c)
    target_link_libraries(c_api_smoke PRIVATE docscan)
    add_test(NAME c_api_smoke COMMAND c_api_smoke)
    if(WIN32)
        # Isolated fake older runtime: ensure incompatible host DLLs return an error, not a null API crash.
        add_library(incompatible_runtime SHARED tests/incompatible_runtime.c tests/incompatible_runtime.def)
        target_include_directories(incompatible_runtime PRIVATE ${ORT_INCLUDE})
        set_target_properties(incompatible_runtime PROPERTIES OUTPUT_NAME onnxruntime
            RUNTIME_OUTPUT_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/incompatible-runtime/$<CONFIG>"
            ARCHIVE_OUTPUT_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/incompatible-runtime/$<CONFIG>")
        add_executable(runtime_contract tests/runtime_contract.c)
        target_link_libraries(runtime_contract PRIVATE docscan)
        set_target_properties(runtime_contract PROPERTIES
            RUNTIME_OUTPUT_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/incompatible-runtime/$<CONFIG>")
        add_dependencies(runtime_contract incompatible_runtime)
        add_custom_command(TARGET runtime_contract POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different
            $<TARGET_FILE:docscan> $<TARGET_FILE:opencv_core> $<TARGET_FILE:opencv_imgproc>
            $<TARGET_FILE:opencv_imgcodecs> $<TARGET_FILE_DIR:runtime_contract>)
        add_test(NAME runtime_contract COMMAND runtime_contract)
    endif()
    add_executable(pixel_contract tests/pixel_contract.cpp src/geometry.cpp src/filters.cpp src/paper-restoration.cpp src/paper-field.cpp)
    target_include_directories(pixel_contract PRIVATE include src ${ORT_INCLUDE} ${OpenCV_INCLUDE_DIRS})
    target_compile_definitions(pixel_contract PRIVATE ORT_API_MANUAL_INIT)
    target_link_libraries(pixel_contract PRIVATE ${OpenCV_LIBS})
    add_test(NAME pixel_contract COMMAND pixel_contract "${CMAKE_CURRENT_SOURCE_DIR}/tests/golden")
    add_executable(opencv_contract tests/opencv_contract.cpp src/geometry.cpp src/quality.cpp src/opencv-raster.cpp src/paper-restoration.cpp src/paper-field.cpp)
    target_include_directories(opencv_contract PRIVATE include src ${OpenCV_INCLUDE_DIRS})
    target_link_libraries(opencv_contract PRIVATE ${OpenCV_LIBS} docscan)
    add_test(NAME opencv_contract COMMAND opencv_contract)
    add_executable(adaptive_contract tests/adaptive_contract.cpp src/adaptive-enhancement.cpp)
    target_include_directories(adaptive_contract PRIVATE include src ${OpenCV_INCLUDE_DIRS})
    target_link_libraries(adaptive_contract PRIVATE ${OpenCV_LIBS})
    add_test(NAME adaptive_contract COMMAND adaptive_contract "${CMAKE_CURRENT_SOURCE_DIR}/tests/golden/adaptive-v1")
    add_executable(shadow_contract tests/shadow_contract.cpp)
    target_include_directories(shadow_contract PRIVATE ${OpenCV_INCLUDE_DIRS})
    target_link_libraries(shadow_contract PRIVATE docscan ${OpenCV_LIBS})
    add_test(NAME shadow_contract COMMAND shadow_contract "${CMAKE_CURRENT_SOURCE_DIR}/models")
    add_executable(paper_contract tests/paper_contract.cpp src/paper-restoration.cpp src/paper-field.cpp)
    target_include_directories(paper_contract PRIVATE include src ${OpenCV_INCLUDE_DIRS})
    target_link_libraries(paper_contract PRIVATE ${OpenCV_LIBS})
    add_test(NAME paper_contract COMMAND paper_contract "${CMAKE_CURRENT_SOURCE_DIR}/tests/golden/paper-v2")
    # Optional independent parser/rasterizer checks, never required by the runtime library.
    find_package(Python3 COMPONENTS Interpreter)
    if(Python3_Interpreter_FOUND)
        execute_process(COMMAND "${Python3_EXECUTABLE}" -c "import numpy,PIL,pypdf,pypdfium2"
            RESULT_VARIABLE DOCSCAN_PYTHON_TEST_DEPS OUTPUT_QUIET ERROR_QUIET)
        if(DOCSCAN_PYTHON_TEST_DEPS EQUAL 0)
            add_test(NAME integration COMMAND "${Python3_EXECUTABLE}"
                "${CMAKE_CURRENT_SOURCE_DIR}/tests/integration.py" $<TARGET_FILE:docscan>)

        else()
            message(STATUS "Independent integration tests need test-only requirements-test.txt")
        endif()
    endif()
endif()
install(TARGETS docscan EXPORT docscanTargets RUNTIME DESTINATION bin LIBRARY DESTINATION lib ARCHIVE DESTINATION lib)
install(DIRECTORY include/ DESTINATION include)
install(DIRECTORY models/ DESTINATION share/docscan/models)
install(DIRECTORY licenses/ DESTINATION share/docscan/licenses)
install(FILES docs/cpp-library-integration.md docs/adaptive-enhancement.md docs/adaptive-enhancement-results.json docs/pretrained-shadow.md README.md docs/architecture.md docs/wasm.md docs/opencv.md docs/verification.md docs/dependencies.md docs/corner-refinement.md docs/corner-refinement-results.json docs/benchmark-results.json docs/paper-restoration-v1.md docs/paper-restoration-v1-results.json docs/paper-restoration-v1-holdout.json docs/paper-restoration-v1-real-corpus.json docs/paper-restoration-v2.md docs/paper-restoration-v2-results.json docs/paper-restoration-v2-holdout.json docs/paper-restoration-v2-real-corpus.json DESTINATION share/docscan/docs)
install(FILES examples/docscan.py examples/docscan_sdk.py examples/scan.cpp DESTINATION share/docscan/examples)
install(EXPORT docscanTargets NAMESPACE docscan:: DESTINATION lib/cmake/docscan)
configure_package_config_file(cmake/docscanConfig.cmake.in "${CMAKE_CURRENT_BINARY_DIR}/docscanConfig.cmake"
    INSTALL_DESTINATION lib/cmake/docscan)
write_basic_package_version_file("${CMAKE_CURRENT_BINARY_DIR}/docscanConfigVersion.cmake"
    VERSION ${PROJECT_VERSION} COMPATIBILITY SameMajorVersion)
install(FILES "${CMAKE_CURRENT_BINARY_DIR}/docscanConfig.cmake" "${CMAKE_CURRENT_BINARY_DIR}/docscanConfigVersion.cmake"
    DESTINATION lib/cmake/docscan)
if(WIN32)
    install(FILES $<TARGET_RUNTIME_DLLS:docscan> DESTINATION bin)
endif()
